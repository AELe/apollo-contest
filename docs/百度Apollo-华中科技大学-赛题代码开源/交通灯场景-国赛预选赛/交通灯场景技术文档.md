# 交通灯场景技术文档

## 一、场景回放与赛题分析

**核心需求**：

1. 主车在监测到红灯信号后，必须在停止线前**1.5-2.0m** 内精确停车
2. 车辆在红灯状态下可红灯右转
3. 车辆右转至直行车道过程中，若遇到前方车道有停驶车辆，主车需要借道绕行
4. 主车与障碍物的横向距离应至少保持1米，借道避障时车速不得超过5 m/s

**评判关注点（示例）**：

| 情况                 | 关注点/结果                |
| -------------------- | ------------------------- |
| 停止位置不在1.5-2.0m内        | 扣20分         |
| 车辆与障碍物横向距离小于1m        | 扣20分              |
| 车辆避障时速度大于5m/s       | 每超速1m/s，每帧扣1分             |

**技术难点**：

- 停止距离精度控制（±0.5m）
- 红灯右转判定
- 避障时横向距离与速度控制

**解决思路（总览）**：

1. 参数调优：
    - 设置stop_sign/conf/default_conf.pb.txt的stop_distance
    - 提高lane_follow_path/conf/default_conf.pb.txt下l_weight的权重
    - 设置planning_component/conf/planning.conf下的obstacle_lat_buffer
2. 代码增强：
   - 交通路口前增加限速区域，保证可控刹停
   - 放宽右转场景切入条件（引入曲率意图识别）
   - 借道完成前不结束右转阶段

**右转场景可转移条件放宽**

```cpp
// 判断是否可转移到无保护右转场景
bool TrafficLightUnprotectedRightTurnScenario::IsTransferable(
        const Scenario* const other_scenario,
        const Frame& frame) {
    // 必须处于车道跟随模式，否则不进入该场景
    if (!frame.local_view().planning_command->has_lane_follow_command()) {
        AINFO << "no lane_follow command";
        return false;
    }

    // 获取参考线与首个遇到的路口重叠信息
    const auto& reference_line_info = frame.reference_line_info().front();
    const auto& first_encountered_overlaps = reference_line_info.FirstEncounteredOverlaps();
    hdmap::PathOverlap* traffic_sign_overlap = nullptr;
    for (const auto& overlap : first_encountered_overlaps) {
        // 遇到停止/让行标志则不进入红灯右转场景
        if (overlap.first == ReferenceLineInfo::STOP_SIGN || overlap.first == ReferenceLineInfo::YIELD_SIGN) {
            AINFO << "first encountered overlap is stop sign or yield sign";
            return false;
        } else if (overlap.first == ReferenceLineInfo::SIGNAL) {
            // 记录第一个信号灯重叠
            traffic_sign_overlap = const_cast<hdmap::PathOverlap*>(&overlap.second);
            break;
        }
    }
    // 未找到信号灯重叠则不进入
    if (traffic_sign_overlap == nullptr) {
        AINFO << "no traffic light overlap";
        return false;
    }
    // 基于距离将同组信号灯合并
    const std::vector<hdmap::PathOverlap>& traffic_light_overlaps
            = reference_line_info.reference_line().map_path().signal_overlaps();
    const double start_check_distance
            = context_.scenario_config.start_traffic_light_scenario_distance() + 5.0;  // 增加5米检测距离
    const double adc_front_edge_s = reference_line_info.AdcSlBoundary().end_s();
    // 查找与第一个信号灯同组的所有信号灯
    std::vector<hdmap::PathOverlap> next_traffic_lights;
    static constexpr double kTrafficLightGroupingMaxDist = 2.0;  // 组内最大间距（m）
    for (const auto& overlap : traffic_light_overlaps) {
        const double dist = overlap.start_s - traffic_sign_overlap->start_s;
        if (fabs(dist) <= kTrafficLightGroupingMaxDist) {
            next_traffic_lights.push_back(overlap);
        }
    }
    bool traffic_light_scenario = false; // 标记是否满足交通灯场景进入条件
    // 遍历同组信号灯，检查距离与灯色
    for (const auto& overlap : next_traffic_lights) {
        const double adc_distance_to_traffic_light = overlap.start_s - adc_front_edge_s;
        ADEBUG << "traffic_light[" << overlap.object_id << "] start_s[" << overlap.start_s
               << "] adc_distance_to_traffic_light[" << adc_distance_to_traffic_light << "]";

        // 基于距离判定是否进入交通灯场景
        if (adc_distance_to_traffic_light <= 0.0 || adc_distance_to_traffic_light > start_check_distance) {
            continue; // 已经过灯或距离过远则跳过
        }

        const auto& signal_color = frame.GetSignal(overlap.object_id).color(); // 获取灯色
        ADEBUG << "traffic_light_id[" << overlap.object_id << "] start_s[" << overlap.start_s << "] color["
               << signal_color << "]";

        if (signal_color == perception::TrafficLight::RED || signal_color == perception::TrafficLight::YELLOW) {
            traffic_light_scenario = true; // 红/黄灯满足进入条件
            break;
        }
    }
    if (!traffic_light_scenario) {
        AINFO << "no traffic light scenario";
        return false;
    }

    // 若不是右转车道，需额外具备明显右转意图（曲率阈值）
    const auto& turn_type = reference_line_info.GetPathTurnType(traffic_sign_overlap->start_s);
    const SLBoundary& adc_sl_boundary = reference_line_info.AdcSlBoundary();

    // 基于参考线的局部曲率判断右转意图
    const auto& AllPoints = reference_line_info.reference_line().GetReferencePoints(
            adc_sl_boundary.end_s(), adc_sl_boundary.end_s() + 10.0);
    bool is_need_right_turn = false;
    for (auto& point : AllPoints) {
        if (point.kappa() < -0.05) { // 右转曲率阈值
            is_need_right_turn = true;
            break;
        }
    }

    // 非右转车道且无明显右转意图则不进入
    if (turn_type != hdmap::Lane::RIGHT_TURN && !is_need_right_turn) {
        return false;
    }

    // 记录当前场景相关的信号灯ID，供后续阶段使用
    context_.current_traffic_light_overlap_ids.clear();
    for (const auto& overlap : next_traffic_lights) {
        context_.current_traffic_light_overlap_ids.push_back(overlap.object_id);
    }
    return true; // 满足条件，允许转移到该场景
}

// 无保护右转场景：路口巡航阶段处理
StageResult TrafficLightUnprotectedRightTurnStageIntersectionCruise::Process(
        const common::TrajectoryPoint& planning_init_point,
        Frame* frame) {
    ADEBUG << "stage: IntersectionCruise";
    CHECK_NOTNULL(frame);

    // 计算目标终点的SL坐标与车辆当前位置，用于借道判定
    common::SLPoint dest_sl;                                      // 目标终点在参考线下的SL坐标
    const auto routing_end = frame->local_view().end_lane_way_point; // 路由终点 WayPoint
    const auto& reference_line_info = frame->reference_line_info().front(); // 当前参考线信息
    const auto& reference_line = reference_line_info.reference_line();      // 参考线对象
    reference_line.XYToSL(routing_end->pose(), &dest_sl);         // 坐标转换为SL

    const SLBoundary& adc_sl_boundary = reference_line_info.AdcSlBoundary(); // 车辆SL边界
    const double adc_center_edge_l = adc_sl_boundary.end_l() + adc_sl_boundary.start_l(); // 近似车辆横向中心

    bool is_borrow = false; // 是否处于借道状态

    double blocking_obstacle_end_s = 0.0;   // 前方静态障碍物尾点s
    double blocking_obstacle_start_s = 0.0; // 前方静态障碍物首点s
    const auto& path_decider_status = injector_->planning_context()->planning_status().path_decider();
    // 获取前方静态障碍物的ID
    const std::string blocking_obstacle_id = path_decider_status.front_static_obstacle_id();
    // 如果存在前方静态障碍物，则查询其SL边界
    if (!blocking_obstacle_id.empty()) {
        const Obstacle* blocking_obstacle = reference_line_info.path_decision().obstacles().Find(blocking_obstacle_id); // 查找障碍物
        if (blocking_obstacle != nullptr) {
            blocking_obstacle_end_s = blocking_obstacle->PerceptionSLBoundary().end_s();   // 障碍物尾点s
            blocking_obstacle_start_s = blocking_obstacle->PerceptionSLBoundary().start_s(); // 障碍物首点s
        }
    }

    // 车辆与障碍物的s距离较近，认为在借道上下文中
    if (fabs(adc_sl_boundary.start_s() - blocking_obstacle_start_s) < 25.0) {
        is_borrow = true;
    }

    // 车辆横向中心与目标终点横向差值较大，认为仍在借道
    if (fabs(adc_center_edge_l - dest_sl.l()) > 0.30) {
        is_borrow = true;
    }

    // 在障碍物前20m到后60m范围内设置5m/s限速
    ReferenceLine* reference_line_ptr = frame->mutable_reference_line_info()->front().mutable_reference_line();
    reference_line_ptr->AddSpeedLimit(blocking_obstacle_start_s - 20.0, blocking_obstacle_end_s + 60.0, 5.0);

    // 执行参考线上的各项规划任务
    StageResult result = ExecuteTaskOnReferenceLine(planning_init_point, frame);
    if (result.HasError()) {
        AERROR << "TrafficLightUnprotectedRightTurnStageIntersectionCruise "
               << "plan error";
    }

    // 检查是否满足阶段结束条件（但借道未完成则继续）
    bool stage_done = CheckDone(*frame, injector_->planning_context(), false);

    if (stage_done && !is_borrow) {
        return FinishStage(); // 借道完成后允许结束该阶段
    }
    return result.SetStageStatus(StageStatusType::RUNNING); // 否则持续运行
}
```
